import numpy as np
import random
import rospy
from geometry_msgs.msg import Twist
import sys, select, termios, tty
from nav_msgs.msg import Odometry
from sensor_msgs.msg import LaserScan
from squaternion import Quaternion
import csv
import threading
import time
import uuid

class ParticleFilter:
    def __init__(self, num_particles, initial_state, motion_noise, measurement_noise):
        self.num_particles = num_particles
        self.particles = np.tile(initial_state, (num_particles, 1))
        self.weights = np.ones(num_particles) / num_particles
        self.base_motion_noise = motion_noise
        self.base_measurement_noise = measurement_noise
        self.motion_noise = motion_noise
        self.measurement_noise = measurement_noise
        self.last_measurement = initial_state.copy()
        self.anomaly_count = 0
        self.std_threshold = 0.8  
        self.last_odom_vel = np.array([0.0, 0.0, 0.0])
        self.anomaly_timer = 0.0  
        self.anomaly_persistence_duration = 2.0  # Seconds

    def predict(self, control_input, dt=0.1, attack_freq=0):
        v, w = control_input[0], control_input[2]
        theta = self.particles[:, 2]

        if attack_freq >= 50:  
            v, w = self.last_odom_vel[0] * 0.7, self.last_odom_vel[2] * 0.7  
            self.motion_noise = self.base_motion_noise * 3  
        else:
            self.motion_noise = self.base_motion_noise

        noise = np.random.normal(0, self.motion_noise * max(abs(v), abs(w), 0.1), (self.num_particles, 3))
        self.particles[:, 0] += v * np.cos(theta) * dt + noise[:, 0]
        self.particles[:, 1] += v * np.sin(theta) * dt + noise[:, 1] * 0.05
        self.particles[:, 2] += w * dt + noise[:, 2]

    def update(self, measurement, lidar_ranges=None, attack_freq=0):
        self.measurement_noise = self.base_measurement_noise * 1.5 if attack_freq >= 50 else self.base_measurement_noise

        for i in range(self.num_particles):
            diff = self.particles[i] - measurement
            weight = np.exp(-np.sum(diff**2) / (2 * self.measurement_noise**2))
            if lidar_ranges is not None and len(lidar_ranges) > 0:
                expected_range = 8.0  # Simplified environment assumption
                lidar_diff = abs(lidar_ranges[len(lidar_ranges)//2] - expected_range) if len(lidar_ranges) > 10 else 0.0
                weight *= np.exp(-lidar_diff**2 / (2 * 0.4**2))
            self.weights[i] = weight
        self.weights += 1e-300
        self.weights /= np.sum(self.weights)

    def resample(self, attack_freq=0):
        indices = np.random.choice(range(self.num_particles), size=self.num_particles, p=self.weights)
        self.particles = self.particles[indices]
        self.weights.fill(1.0 / self.num_particles)
        particle_std = self.get_particle_std()
        if attack_freq < 50 and np.any(particle_std > self.std_threshold):
            self.reset_particles(self.last_measurement)

    def reset_particles(self, measurement):
        self.particles = np.tile(measurement, (self.num_particles, 1)) + \
                         np.random.normal(0, 0.05, (self.num_particles, 3))
        self.weights.fill(1.0 / self.num_particles)
        rospy.loginfo("Particles reset due to excessive drift.")

    def estimate(self):
        return np.average(self.particles, weights=self.weights, axis=0)

    def get_particle_std(self):
        return np.std(self.particles, axis=0)

    def detect_anomaly(self, measurement, attack_freq=0, control_input=np.array([0.0, 0.0, 0.0]), dt=0.1):
        estimated_state = self.estimate()
        diff = np.linalg.norm(estimated_state - measurement)
        rate_of_change = np.linalg.norm(measurement - self.last_measurement) / dt
        expected_change = np.linalg.norm(control_input * dt)
        particle_std = self.get_particle_std()

        self.last_odom_vel = (measurement - self.last_measurement) / dt
        self.last_measurement = measurement.copy()

        is_anomaly = False
        if attack_freq >= 50:
            is_anomaly = True
            self.anomaly_count += 1
            self.anomaly_timer = self.anomaly_persistence_duration
        elif diff > 4.0 or np.any(particle_std > self.std_threshold):
            is_anomaly = True
            self.reset_particles(measurement)
            self.anomaly_timer = self.anomaly_persistence_duration

        if self.anomaly_timer > 0:
            is_anomaly = True
            self.anomaly_timer -= dt

        return {
            'is_anomaly': is_anomaly,
            'diff': diff,
            'rate_of_change': rate_of_change,
            'expected_change': expected_change,
            'estimated_state': estimated_state,
            'particle_std_x': particle_std[0],
            'particle_std_y': particle_std[1],
            'particle_std_theta': particle_std[2]
        }

class Teleop:
    def __init__(self):
        rospy.init_node('robot_teleop', anonymous=True)

        self.odom_x = self.odom_y = self.odom_z = self.angle = 0.0
        self.lidar_data = []
        self.linear_vel = self.angular_vel = 0.0
        self.attack_active = False
        self.attack_frequency = 50  # Attack frequency in Hz
        self.attack_interval = 1.0 / self.attack_frequency  # Time interval between attacks

        self.data_lock = threading.Lock()
        self.output_data = [['time', 'odom_x', 'odom_y', 'odom_theta', 
                           'est_x', 'est_y', 'est_theta', 
                           'diff', 'rate_of_change', 'expected_change',
                           'particle_std_x', 'particle_std_y', 'particle_std_theta',
                           'linear_vel', 'angular_vel', 
                           'anomaly_detected', 'attack_active']]

        self.pub_vel = rospy.Publisher('cmd_vel', Twist, queue_size=10)
        self.sub_odom = rospy.Subscriber('odometry/filtered_map', Odometry, self.odom_callback)
        self.sub_lidar = rospy.Subscriber('scan', LaserScan, self.lidar_callback)
        self.rate = rospy.Rate(10)

        self.settings = termios.tcgetattr(sys.stdin)

        initial_state = np.array([0.0, 0.0, 0.0])
        self.particle_filter = ParticleFilter(num_particles=800, 
                                            initial_state=initial_state, 
                                            motion_noise=0.08, 
                                            measurement_noise=0.08)

        self.last_time = time.time()
        self.control_input = np.array([0.0, 0.0, 0.0])
        rospy.on_shutdown(self.save_data)

    def odom_callback(self, msg):
        self.odom_x = msg.pose.pose.position.x
        self.odom_y = msg.pose.pose.position.y
        q = Quaternion(msg.pose.pose.orientation.w,
                      msg.pose.pose.orientation.x,
                      msg.pose.pose.orientation.y,
                      msg.pose.pose.orientation.z)
        self.angle = round(q.to_euler(degrees=False)[2], 4)

        measurement = np.array([self.odom_x, self.odom_y, self.angle])
        dt = time.time() - self.last_time
        attack_freq = 50 if self.attack_active else 0
        self.particle_filter.predict(self.control_input, dt, attack_freq)
        self.particle_filter.update(measurement, self.lidar_data, attack_freq)
        self.particle_filter.resample(attack_freq)

        anomaly_info = self.particle_filter.detect_anomaly(measurement, attack_freq, self.control_input, dt)
        self.last_time = time.time()

        print(f"\nTime: {rospy.get_time():.2f}")
        print(f"Odometry: x={self.odom_x:.2f}, y={self.odom_y:.2f}, theta={self.angle:.2f}")
        print(f"Estimated: x={anomaly_info['estimated_state'][0]:.2f}, "
              f"y={anomaly_info['estimated_state'][1]:.2f}, "
              f"theta={anomaly_info['estimated_state'][2]:.2f}")
        print(f"Diff: {anomaly_info['diff']:.3f}, Rate: {anomaly_info['rate_of_change']:.2f}, "
              f"Expected: {anomaly_info['expected_change']:.2f}")
        print(f"Particle Std: x={anomaly_info['particle_std_x']:.3f}, "
              f"y={anomaly_info['particle_std_y']:.3f}, "
              f"theta={anomaly_info['particle_std_theta']:.3f}")
        print(f"Anomaly: {anomaly_info['is_anomaly']}, Attack: {self.attack_active}")

        if anomaly_info['is_anomaly']:
            rospy.logwarn(f"Anomaly detected! Diff: {anomaly_info['diff']:.3f}")

        with self.data_lock:
            self.output_data.append([
                rospy.get_time(),
                self.odom_x, self.odom_y, self.angle,
                anomaly_info['estimated_state'][0],
                anomaly_info['estimated_state'][1],
                anomaly_info['estimated_state'][2],
                anomaly_info['diff'],
                anomaly_info['rate_of_change'],
                anomaly_info['expected_change'],
                anomaly_info['particle_std_x'],
                anomaly_info['particle_std_y'],
                anomaly_info['particle_std_theta'],
                self.linear_vel, self.angular_vel,
                int(anomaly_info['is_anomaly']),
                int(self.attack_active)
            ])

    def lidar_callback(self, msg):
        self.lidar_data = msg.ranges

    def publish_velocity(self):
        cmd_vel = Twist()
        cmd_vel.linear.x = self.linear_vel
        cmd_vel.angular.z = self.angular_vel
        self.pub_vel.publish(cmd_vel)

    def attack_flood(self):
        rospy.logwarn("Starting 50Hz attack...")
        last_publish_time = time.time()

        while not rospy.is_shutdown() and self.attack_active:
            current_time = time.time()
            if current_time - last_publish_time >= self.attack_interval:
                msg = Twist()
                msg.linear.x = random.uniform(-1.5, 1.5)
                msg.angular.z = random.uniform(-1.5, 1.5)
                self.pub_vel.publish(msg)
                last_publish_time = current_time
            time.sleep(0.001)  
        rospy.logwarn("Attack stopped.")

    def get_key(self):
        tty.setraw(sys.stdin.fileno())
        rlist, _, _ = select.select([sys.stdin], [], [], 0.1)
        key = sys.stdin.read(1) if rlist else ''
        termios.tcsetattr(sys.stdin, termios.TCSADRAIN, self.settings)
        return key

    def save_data(self):
        filename = f'pf_data_{int(time.time())}.csv'
        with self.data_lock:
            with open(filename, 'w', newline='') as file:
                writer = csv.writer(file)
                writer.writerows(self.output_data)
        rospy.loginfo(f"Data saved to {filename}")

    def main(self):
        print("Controls: w:forward, s:backward, a:left, d:right, "
              "space:stop, t:start attack, y:stop attack, Ctrl+C:quit")

        try:
            while not rospy.is_shutdown():
                key = self.get_key()
                if key == 'w':
                    self.linear_vel = 0.5
                    self.angular_vel = 0.0
                elif key == 's':
                    self.linear_vel = -0.5
                    self.angular_vel = 0.0
                elif key == 'a':
                    self.linear_vel = 0.0
                    self.angular_vel = 0.5
                elif key == 'd':
                    self.linear_vel = 0.0
                    self.angular_vel = -0.5
                elif key == ' ':
                    self.linear_vel = 0.0
                    self.angular_vel = 0.0
                elif key == 't' and not self.attack_active:
                    self.attack_active = True
                    threading.Thread(target=self.attack_flood, daemon=True).start()
                elif key == 'y':
                    self.attack_active = False
                elif key == '\x03':
                    break

                self.control_input = np.array([self.linear_vel, 0.0, self.angular_vel])
                self.publish_velocity()
                self.rate.sleep()

        except Exception as e:
            print(f"Error: {e}")
        finally:
            self.attack_active = False
            self.linear_vel = 0.0
            self.angular_vel = 0.0
            self.publish_velocity()
            termios.tcsetattr(sys.stdin, termios.TCSADRAIN, self.settings)

if __name__ == '__main__':
    try:
        teleop = Teleop()
        teleop.main()
    except rospy.ROSInterruptException:
        print("ROS node interrupted.")
