class UnscentedKalmanFilter:
    OVERFLOW_LIDAR_THRESHOLD = 10_000
    P_FLOOR                  = 1e-6
    MAX_CONSECUTIVE_REPAIRS  = 10

    def __init__(self, initial_state, process_noise, measurement_noise):
        self.state = np.array([initial_state[0], initial_state[1], initial_state[2],
                               0.0, 0.0, 0.0], dtype=float)
        self.n     = len(self.state)

        self.P0     = np.eye(self.n) * 0.1
        self.P      = self.P0.copy()
        self.Q      = np.eye(self.n) * process_noise
        self.R      = np.eye(5)      * measurement_noise
        self.base_Q = self.Q.copy()
        self.base_R = self.R.copy()

        self.last_measurement    = np.array([initial_state[0], initial_state[1],
                                             initial_state[2], 0.0, 0.0])
        self.anomaly_count       = 0
        self.error_threshold     = 1.0
        self._repair_count       = 0

        self.alpha   = 1e-3
        self.beta    = 2
        self.kappa   = 0
        self.lambda_ = self.alpha**2 * (self.n + self.kappa) - self.n

        self.Wm    = np.full(2 * self.n + 1, 1.0 / (2.0 * (self.n + self.lambda_)))
        self.Wc    = np.full(2 * self.n + 1, 1.0 / (2.0 * (self.n + self.lambda_)))
        self.Wm[0] = self.lambda_ / (self.n + self.lambda_)
        self.Wc[0] = self.lambda_ / (self.n + self.lambda_) + (1.0 - self.alpha**2 + self.beta)

        self._odom_history       = []
        self._history_maxlen     = 30
        self._last_msg_stamp     = None

    def _symmetrise(self, M):
        return 0.5 * (M + M.T)

    def _repair_covariance(self):
        self._repair_count += 1
        if self._repair_count > self.MAX_CONSECUTIVE_REPAIRS:
            rospy.logwarn("[UKF] Covariance diverged – resetting P to P0")
            self.P = self.P0.copy()
            self._repair_count = 0
            return

        self.P = self._symmetrise(self.P)
        eigvals, eigvecs = np.linalg.eigh(self.P)
        eigvals = np.maximum(eigvals, self.P_FLOOR)
        self.P  = eigvecs @ np.diag(eigvals) @ eigvecs.T
        self.P  = self._symmetrise(self.P)

    def _cholesky_robust(self, M):
        M = self._symmetrise(M)

        try:
            return np.linalg.cholesky(M)
        except np.linalg.LinAlgError:
            pass

        for eps in [1e-6, 1e-4, 1e-2, 1e-1]:
            try:
                return np.linalg.cholesky(M + np.eye(self.n) * eps)
            except np.linalg.LinAlgError:
                pass

        eigvals, eigvecs = np.linalg.eigh(M)
        eigvals = np.maximum(eigvals, self.P_FLOOR)
        M_repaired = eigvecs @ np.diag(eigvals) @ eigvecs.T
        M_repaired = self._symmetrise(M_repaired)
        try:
            return np.linalg.cholesky(M_repaired)
        except np.linalg.LinAlgError:
            raise RuntimeError("[UKF] _cholesky_robust: all attempts failed")

    def motion_model(self, state, control_input, dt):
        x, y, theta, vx, vy, omega = state
        v, w = control_input[0], control_input[1]
        return np.array([
            x + vx * dt * np.cos(theta),
            y + vy * dt * np.sin(theta),
            theta + omega * dt,
            v,
            0.0,
            w
        ], dtype=float)

    def measurement_model(self, state):
        return np.array([state[0], state[1], state[2], state[0], state[1]])

    def generate_sigma_points(self):
        mat    = (self.n + self.lambda_) * self.P
        sqrt_P = self._cholesky_robust(mat)

        sigma    = np.zeros((2 * self.n + 1, self.n))
        sigma[0] = self.state
        for i in range(self.n):
            sigma[i + 1]          = self.state + sqrt_P[i]
            sigma[self.n + i + 1] = self.state - sqrt_P[i]
        return sigma

    def predict(self, control_input, dt=0.1, attack_mode='normal'):
        if attack_mode in ('flood', 'overflow'):
            self.Q = self.base_Q * 4.0
        elif attack_mode == 'replay':
            self.Q = self.base_Q * 2.0
        else:
            self.Q = self.base_Q

        sigma      = self.generate_sigma_points()
        sigma_pred = np.array([self.motion_model(pt, control_input, dt)
                               for pt in sigma])

        self.state = np.sum(self.Wm[:, None] * sigma_pred, axis=0)
        diff       = sigma_pred - self.state
        P_pred     = np.sum(self.Wc[:, None, None]
                            * np.einsum('ij,ik->ijk', diff, diff), axis=0) + self.Q

        self.P = self._symmetrise(P_pred)

        if np.any(np.linalg.eigvalsh(self.P) < 0):
            self._repair_covariance()
        else:
            self._repair_count = 0

    def update(self, measurement, attack_mode='normal'):
        sigma    = self.generate_sigma_points()
        meas_sig = np.array([self.measurement_model(pt) for pt in sigma])
        z_pred   = np.sum(self.Wm[:, None] * meas_sig, axis=0)

        diff_z = meas_sig - z_pred
        diff_x = sigma - self.state

        if attack_mode == 'flood':
            self.R = np.diag([2.0, 2.0, 2.0, 0.5, 0.5])
        elif attack_mode == 'overflow':
            self.R = np.diag([1.5, 1.5, 1.5, 0.3, 0.3])
        elif attack_mode == 'replay':
            self.R = np.diag([5.0, 5.0, 5.0, 0.5, 0.5])
        else:
            self.R = self.base_R

        S   = (np.sum(self.Wc[:, None, None]
                      * np.einsum('ij,ik->ijk', diff_z, diff_z), axis=0)
               + self.R)
        Pxz = np.sum(self.Wc[:, None, None]
                     * np.einsum('ij,ik->ijk', diff_x, diff_z), axis=0)

        S  = self._symmetrise(S)
        K  = Pxz @ np.linalg.inv(S)

        z  = np.array(measurement)
        y  = z - z_pred
        self.state = self.state + K @ y

        m = len(z)
        H = np.zeros((m, self.n))
        H[0, 0] = H[1, 1] = H[2, 2] = H[3, 0] = H[4, 1] = 1.0
        IKH  = np.eye(self.n) - K @ H
        self.P = IKH @ self.P @ IKH.T + K @ self.R @ K.T
        self.P = self._symmetrise(self.P)

        if np.any(np.linalg.eigvalsh(self.P) < 0):
            self._repair_covariance()

        return y

    def estimate(self):
        return self.state[:3]

    def _update_odom_history(self, measurement, timestamp=None):
        entry = {
            'pos':   np.array(measurement[:3]),
            'time':  timestamp if timestamp is not None else time.time()
        }
        self._odom_history.append(entry)
        if len(self._odom_history) > self._history_maxlen:
            self._odom_history.pop(0)

    def _replay_pattern_detected(self, control_input):
        if len(self._odom_history) < 10:
            return False

        recent   = np.array([e['pos'] for e in self._odom_history[-15:]])
        pos_var  = np.var(recent[:, :2], axis=0).mean()
        head_var = np.var(recent[:, 2])

        moving   = np.linalg.norm(control_input) > 0.05
        turning  = abs(control_input[2]) > 0.05

        pos_frozen = moving and pos_var < 1e-4
        head_frozen = turning and head_var < 1e-5

        stamp_looping = False
        if len(self._odom_history) >= 5:
            stamps = [e['time'] for e in self._odom_history[-5:]]
            stamp_diffs = np.diff(stamps)
            if np.std(stamp_diffs) < 1e-4 and np.mean(stamp_diffs) < 0.01:
                stamp_looping = True

        return pos_frozen or head_frozen or stamp_looping

    def detect_anomaly(self, measurement, attack_mode='normal',
                       control_input=np.array([0.0, 0.0, 0.0]), dt=0.1,
                       msg_stamp=None):
        residual        = self.update(measurement, attack_mode)
        diff            = np.linalg.norm(residual)
        rate_of_change  = np.linalg.norm(
            np.array(measurement) - self.last_measurement) / max(dt, 1e-6)
        expected_change = np.linalg.norm(control_input * dt)

        self._update_odom_history(measurement, timestamp=msg_stamp)
        self.last_measurement = np.array(measurement).copy()

        is_anomaly = False

        if attack_mode == 'flood':
            is_anomaly = (diff > self.error_threshold or
                          abs(rate_of_change - expected_change) > 1.0)
            if is_anomaly:
                self.anomaly_count += 1

        elif attack_mode == 'overflow':
            is_anomaly = diff > self.error_threshold * 1.5
            if is_anomaly:
                self.anomaly_count += 1

        elif attack_mode == 'replay':
            replay_detected = self._replay_pattern_detected(control_input)
            is_anomaly      = replay_detected or diff > 2.0
            if is_anomaly:
                self.anomaly_count += 1
            if replay_detected:
                rospy.logwarn("[UKF] Replay pattern confirmed by multi-criterion detector")

        elif diff > 2.0:
            is_anomaly = True

        return {
            'is_anomaly':       is_anomaly,
            'diff':             diff,
            'rate_of_change':   rate_of_change,
            'expected_change':  expected_change,
            'estimated_state':  self.estimate(),
            'covariance_trace': np.trace(self.P[:3, :3])
        }

class Teleop:
    def __init__(self):
        rospy.init_node('husky_teleoperation_ukf_lidar')

        self.odom_x = self.odom_y = self.odom_z = self.angle = 0.0
        self.lidar_x     = self.lidar_y   = 0.0
        self.lidar_front = float('inf')
        self.lidar_data  = []
        self.linear_vel  = self.angular_vel = 0.0

        self.flood_active    = False
        self.overflow_active = False
        self.replay_active   = False
        self.recorded_msgs   = []
        self.lidar_overflow  = False

        self.data_lock   = threading.Lock()
        self.output_data = [[
            'time', 'attack_type',
            'odom_x', 'odom_y', 'odom_theta',
            'lidar_x', 'lidar_y', 'lidar_front',
            'est_x', 'est_y', 'est_theta',
            'diff', 'rate_of_change', 'expected_change',
            'covariance_trace',
            'linear_vel', 'angular_vel',
            'anomaly_detected', 'attack_active'
        ]]

        self.pub_vel        = rospy.Publisher('cmd_vel', Twist, queue_size=10)
        self.sub_odom       = rospy.Subscriber('odometry/filtered', Odometry,
                                               self.odom_callback)
        self.sub_lidar      = rospy.Subscriber('front/scan', LaserScan,
                                               self.lidar_callback)
        self.sub_pointcloud = rospy.Subscriber('points', PointCloud2,
                                               self.pointcloud_callback)
        self.rate           = rospy.Rate(10)
        self.settings       = termios.tcgetattr(sys.stdin)

        initial_state = np.array([0.0, 0.0, 0.0])
        self.ukf      = UnscentedKalmanFilter(initial_state=initial_state,
                                              process_noise=0.01,
                                              measurement_noise=0.1)
        self.last_time     = time.time()
        self.control_input = np.array([0.0, 0.0, 0.0])
        rospy.on_shutdown(self.save_data)

    def get_attack_mode(self):
        if self.replay_active:   return 'replay'
        if self.flood_active:    return 'flood'
        if self.overflow_active: return 'overflow'
        return 'normal'

    def odom_callback(self, msg):
        self.odom_x = msg.pose.pose.position.x
        self.odom_y = msg.pose.pose.position.y
        self.odom_z = msg.pose.pose.position.z
        q = Quaternion(msg.pose.pose.orientation.w,
                       msg.pose.pose.orientation.x,
                       msg.pose.pose.orientation.y,
                       msg.pose.pose.orientation.z)
        self.angle = round(q.to_euler(degrees=False)[2], 4)

        if not self.replay_active:
            self.recorded_msgs.append(msg)

        measurement = [self.odom_x, self.odom_y, self.angle,
                       self.lidar_x, self.lidar_y]
        dt          = max(time.time() - self.last_time, 1e-4)
        attack_mode = self.get_attack_mode()

        try:
            msg_stamp = msg.header.stamp.to_sec()
        except Exception:
            msg_stamp = None

        try:
            self.ukf.predict(self.control_input, dt, attack_mode)
            anomaly_info = self.ukf.detect_anomaly(
                measurement, attack_mode, self.control_input, dt,
                msg_stamp=msg_stamp)
        except Exception as e:
            rospy.logerr(f"[UKF] Filter step failed: {e} — resetting P")
            self.ukf.P = self.ukf.P0.copy()
            self.last_time = time.time()
            return

        self.last_time = time.time()

        if anomaly_info['is_anomaly'] and attack_mode in ('flood', 'overflow', 'replay'):
            self.correct_control_input(anomaly_info['estimated_state'])

        print(f"\nOdometry : x={self.odom_x:.2f}, y={self.odom_y:.2f}, "
              f"theta={self.angle:.2f}")
        print(f"LiDAR    : x={self.lidar_x:.2f}, y={self.lidar_y:.2f}, "
              f"front={self.lidar_front:.2f}")
        print(f"Estimated: x={anomaly_info['estimated_state'][0]:.2f}, "
              f"y={anomaly_info['estimated_state'][1]:.2f}, "
              f"theta={anomaly_info['estimated_state'][2]:.2f}")
        print(f"Diff: {anomaly_info['diff']:.3f}  Rate: {anomaly_info['rate_of_change']:.2f}  "
              f"Expected: {anomaly_info['expected_change']:.2f}")
        print(f"Cov Trace: {anomaly_info['covariance_trace']:.3f}  "
              f"Anomaly: {anomaly_info['is_anomaly']}  Attack: {attack_mode}")
        print(f"Repair count: {self.ukf._repair_count}")

        if anomaly_info['is_anomaly']:
            rospy.logwarn(f"[UKF] Anomaly! mode={attack_mode}  "
                          f"diff={anomaly_info['diff']:.3f}")

        with self.data_lock:
            self.output_data.append([
                time.time(), attack_mode,
                self.odom_x, self.odom_y, self.angle,
                self.lidar_x, self.lidar_y,
                round(self.lidar_front, 3) if self.lidar_front < 100 else float('inf'),
                anomaly_info['estimated_state'][0],
                anomaly_info['estimated_state'][1],
                anomaly_info['estimated_state'][2],
                anomaly_info['diff'],
                anomaly_info['rate_of_change'],
                anomaly_info['expected_change'],
                anomaly_info['covariance_trace'],
                self.linear_vel, self.angular_vel,
                int(anomaly_info['is_anomaly']),
                int(attack_mode != 'normal')
            ])

    def lidar_callback(self, msg):
        n = len(msg.ranges)
        if n > UnscentedKalmanFilter.OVERFLOW_LIDAR_THRESHOLD:
            rospy.logwarn(f"[UKF] LIDAR overflow detected: {n} ranges")
            self.lidar_overflow = True
            self.lidar_data     = []
            return

        self.lidar_overflow = False
        self.lidar_data     = msg.ranges

        if n > 10:
            val              = msg.ranges[10]
            self.lidar_front = val if val >= 0.01 else float('inf')
        else:
            self.lidar_front = float('inf')

        valid = [r for r in self.lidar_data if r > 0.1]
        if valid:
            min_range    = min(valid)
            self.lidar_x = self.odom_x + min_range * np.cos(self.angle)
            self.lidar_y = self.odom_y + min_range * np.sin(self.angle)

    def pointcloud_callback(self, msg):
        pass

    def publish_velocity(self):
        cmd = Twist()
        cmd.linear.x  = self.linear_vel
        cmd.angular.z = self.angular_vel
        self.pub_vel.publish(cmd)

    def correct_control_input(self, estimated_state):
        self.linear_vel  *= 0.5
        desired_theta     = estimated_state[2]
        self.angular_vel  = float(np.clip(desired_theta - self.angle, -0.5, 0.5))
        rospy.loginfo(f"[UKF] Correcting: lin={self.linear_vel:.2f}  "
                      f"ang={self.angular_vel:.2f}")

    def attack_flood_thread(self):
        rate = rospy.Rate(1000)
        msg  = Twist()
        msg.linear.x  = 1.0
        msg.angular.z = 1.0
        rospy.logwarn("[UKF] Flood attack started (1000 Hz)")
        while self.flood_active and not rospy.is_shutdown():
            self.pub_vel.publish(msg)
            rate.sleep()
        rospy.logwarn("[UKF] Flood attack stopped")

    def attack_overflow_thread(self):
        pub  = rospy.Publisher('front/scan', LaserScan, queue_size=1)
        rate = rospy.Rate(100)
        msg  = LaserScan()
        msg.header.frame_id = "laser"
        msg.ranges          = [1.0] * 1_000_000
        rospy.logwarn("[UKF] Overflow attack started (large LaserScan)")
        while self.overflow_active and not rospy.is_shutdown():
            msg.header.stamp = rospy.Time.now()
            pub.publish(msg)
            rate.sleep()
        rospy.logwarn("[UKF] Overflow attack stopped")

    def attack_replay_thread(self):
        if not self.recorded_msgs:
            rospy.logwarn("[UKF] No messages recorded for replay")
            self.replay_active = False
            return
        pub  = rospy.Publisher('odometry/filtered', Odometry, queue_size=10)
        rate = rospy.Rate(10)
        rospy.logwarn(f"[UKF] Replay attack started ({len(self.recorded_msgs)} msgs)")
        while self.replay_active and not rospy.is_shutdown():
            for msg in self.recorded_msgs:
                if not self.replay_active:
                    break
                pub.publish(msg)
                rate.sleep()
        rospy.logwarn("[UKF] Replay attack stopped")

    def get_key(self):
        tty.setraw(sys.stdin.fileno())
        rlist, _, _ = select.select([sys.stdin], [], [], 0.1)
        key = sys.stdin.read(1) if rlist else ''
        termios.tcsetattr(sys.stdin, termios.TCSADRAIN, self.settings)
        return key

    def save_data(self):
        filename = f'odom_lidar_ukf_data_{int(time.time())}.csv'
        with self.data_lock:
            with open(filename, 'w', newline='') as f:
                csv.writer(f).writerows(self.output_data)
        rospy.loginfo(f"[UKF] Data saved → {filename}")

    def main(self):
        print("=== UKF Teleop (Fixed) – Multi-Attack + Enhanced LIDAR ===")
        print("Movement : w/z/a/d/l/r/f/b/s")
        print("Flood    : t = ON   y = OFF")
        print("Overflow : v = ON   m = OFF")
        print("Replay   : q = ON   e = OFF  (drive normally first)")
        print("Quit     : Ctrl+C\n")

        try:
            while not rospy.is_shutdown():
                key = self.get_key()

                if   key == 'w': self.linear_vel, self.angular_vel =  0.5,  0.5
                elif key == 'z': self.linear_vel, self.angular_vel = -0.5,  0.5
                elif key == 'a': self.linear_vel, self.angular_vel =  0.5, -0.5
                elif key == 'd': self.linear_vel, self.angular_vel = -0.5, -0.5
                elif key == 'l': self.linear_vel, self.angular_vel =  0.0, -0.5
                elif key == 'r': self.linear_vel, self.angular_vel =  0.0,  0.5
                elif key == 'f': self.linear_vel, self.angular_vel =  0.5,  0.0
                elif key == 'b': self.linear_vel, self.angular_vel = -0.5,  0.0
                elif key == 's': self.linear_vel, self.angular_vel =  0.0,  0.0

                elif key == 't' and not self.flood_active:
                    self.flood_active = True
                    threading.Thread(target=self.attack_flood_thread,
                                     daemon=True).start()
                elif key == 'y':
                    self.flood_active = False

                elif key == 'v' and not self.overflow_active:
                    self.overflow_active = True
                    threading.Thread(target=self.attack_overflow_thread,
                                     daemon=True).start()
                elif key == 'm':
                    self.overflow_active = False

                elif key == 'q' and not self.replay_active:
                    if not self.recorded_msgs:
                        print("→ No odometry recorded yet – drive normally first!")
                    else:
                        self.replay_active = True
                        threading.Thread(target=self.attack_replay_thread,
                                         daemon=True).start()
                elif key == 'e':
                    self.replay_active = False

                elif key == '\x03':
                    break

                self.control_input = np.array([self.linear_vel, 0.0, self.angular_vel])
                self.publish_velocity()
                self.rate.sleep()

        except Exception as exc:
            print(exc)
        finally:
            self.flood_active = self.overflow_active = self.replay_active = False
            termios.tcsetattr(sys.stdin, termios.TCSADRAIN, self.settings)

if __name__ == '__main__':
    try:
        Teleop().main()
    except rospy.ROSInterruptException:
        pass
